                      params: dict[str, Any] | None = None) -> Any:
        token = await self.token_provider()
        async with self.http.stream(method, self.base + path,
            headers={"Authorization": "Bearer " + token, "X-Tenant-ID": self.tenant_id},
            json=body, params=params, follow_redirects=False) as response:
            chunks, size = [], 0
            async for chunk in response.aiter_bytes():
                size += len(chunk)
                if size > 5 * 1024 * 1024:
                    raise ValueError("SDK response exceeds 5 MiB")
                chunks.append(chunk)
            import json
            raw = b"".join(chunks)
            retained = httpx.Response(response.status_code, content=raw,
                                      headers=response.headers, request=response.request)
            if not 200 <= response.status_code < 300:
                raise SDKError(retained)
            data = json.loads(raw) if raw else None
        if isinstance(data, dict) and "schema_version" in data and data["schema_version"] != self.schema_version:
            raise ValueError("schema_mismatch: regenerate for the active published schema")
        return data
